    22→    22→/** Serialise un patch : tableaux/objets → JSON, booléens/nombres/texte → brut. */
    23→    23→function serializePatch(patch: Record<string, unknown>, allowed: string[]): Record<string, unknown> {
    24→    24→  const out: Record<string, unknown> = {};
    25→    25→  for (const k of allowed) {
    26→    26→    if (!(k in patch)) continue;
    27→    27→    const v = patch[k];
    28→    28→    out[k] = (Array.isArray(v) || (v && typeof v === 'object')) ? J(v) : v;
    29→    29→  }
    30→    30→  return out;
    31→    31→}

   323→   323→    }
   324→   324→    case 'removeSavedSearch':
   325→   325→      await db.savedSearch.deleteMany({ where: { id: String(p.id), userId: u.id } });
   326→   326→      break;
   327→   327→
   328→   328→    /* ── Consentements (mineurs) ── */
   329→   329→    case 'decideConsent': {
   330→   330→      const c = await db.consentRequest.findUnique({ where: { id: p.id } });
   331→   331→      if (!c || c.guardianId !== u.id) throw new ActionError('Consentement non autorisé.', 403);
   332→   332→      await db.consentRequest.update({ where: { id: c.id }, data: { status: String(p.decision) } });
   333→   333→      await audit(u.name, `Consentement ${c.id} : ${p.decision}`, 'Consentements');
   334→   334→      break;
   335→   335→    }
   336→   336→    case 'requestContact': {
   337→   337→      const pl = await db.player.findUnique({ where: { id: String(p.playerId) } });
   338→   338→      await db.consentRequest.create({ data: {
   339→   339→        id: `cr-${Date.now()}`, guardianId: pl?.guardianId ?? '', playerId: String(p.playerId ?? ''),
   340→   340→        playerName: String(p.playerName ?? ''), requester: u.name, kind: String(p.kind ?? 'Contact'),
   341→   341→        status: 'En attente', date: 'À l\u2019instant',
   342→   342→      } });
   343→   343→      await audit(u.name, `Demande de contact (${p.kind}) sur ${p.playerName}`, 'Consentements');
   344→   344→      break;
   345→   345→    }
   346→   346→
   347→   347→    /* ── Facturation (paiements toujours simulés — aucun encaissement réel) ── */
   348→   348→    case 'checkout': {
   349→   349→      const invoice = `CL-${new Date().getFullYear()}-${Math.floor(1000 + Math.random() * 9000)}`;
   350→   350→      await db.payment.create({ data: {
   351→   351→        id: `pay-${Date.now()}`, userId: u.id, user: u.name, plan: String(p.planName ?? ''),
   352→   352→        amount: String(p.amount ?? ''), method: `${String(p.method ?? '')} (simulé)`,
   353→   353→        status: 'Confirmé', date: today(), invoice,
   354→   354→      } });
   355→   355→      await db.user.update({ where: { id: u.id }, data: {
   356→   356→        subPlanId: String(p.planId ?? 'free'), subStatus: 'Actif',
   357→   357→        subRenews: String(p.renews ?? '30 juin 2027'), subSince: today(),
   358→   358→      } });
   359→   359→      await audit(u.name, `Abonnement ${p.planName} activé (${invoice})`, 'Facturation');
   360→   360→      await notify(u.id, 'Abonnement activé', `${p.planName} est actif — paiement simulé, aucun encaissement réel.`, 'success');
   361→   361→      break;
   362→   362→    }
   363→   363→    case 'cancelSubscription':
   364→   364→      await db.user.update({ where: { id: u.id }, data: { subStatus: 'Expiré', subRenews: '—' } });
   365→   365→      await audit(u.name, 'Résiliation d\u2019abonnement', 'Facturation');
   366→   366→      break;
   367→   367→    case 'updatePlan': {
   368→   368→      if (u.role !== 'admin') throw new ActionError('Réservé à l\u2019administration.', 403);
   369→   369→      await db.plan.update({ where: { id: String(p.id) }, data: serializePatch(p.patch, PLAN_FIELDS) }).catch(() => {});
   370→   370→      await audit(u.name, `Modification du plan ${p.id}`, 'Administration');
   371→   371→      break;
   372→   372→    }
   373→   373→    case 'resolvePayment': {
   374→   374→      if (!isStaff) throw new ActionError('Réservé au support.', 403);
   375→   375→      await db.payment.update({ where: { id: String(p.id) }, data: { status: String(p.status) } }).catch(() => {});
   376→   376→      await audit(u.name, `Paiement ${p.id} → ${p.status}`, 'Facturation');
   377→   377→      break;
   378→   378→    }
   379→   379→
   380→   380→    /* ── Support ── */
   381→   381→    case 'createTicket': {
   382→   382→      await db.ticket.create({ data: {
   383→   383→        id: `t-${Date.now()}`, subject: String(p.subject ?? 'Demande'), requester: u.name,
   384→   384→        topic: String(p.topic ?? ''), status: 'Ouvert', priority: 'Normale', opened: 'À l\u2019instant',
   385→   385→        messages: J([{ from: 'user', body: String(p.body ?? ''), time: 'À l\u2019instant' }]),
   386→   386→      } });
   387→   387→      await audit(u.name, `Nouveau ticket : ${p.subject}`, 'Support');
   388→   388→      break;
   389→   389→    }
   390→   390→    case 'replyTicket': {
   391→   391→      const t = await db.ticket.findUnique({ where: { id: p.id } });
   392→   392→      if (!t || (t.requester !== u.name && !isStaff)) throw new ActionError('Ticket non autorisé.', 403);
   393→   393→      const messages = JSON.parse(t.messages || '[]');
   394→   394→      messages.push({ from: isStaff && t.requester !== u.name ? 'support' : 'user', body: String(p.body ?? ''), time: 'À l\u2019instant' });
   395→   395→      await db.ticket.update({ where: { id: t.id }, data: { messages: J(messages), status: isStaff && t.requester !== u.name ? 'En cours' : t.status } });
   396→   396→      break;
   397→   397→    }
   398→   398→    case 'setTicketStatus': {
   399→   399→      if (!isStaff) throw new ActionError('Réservé au support.', 403);
   400→   400→      await db.ticket.update({ where: { id: String(p.id) }, data: { status: String(p.status) } }).catch(() => {});
   401→   401→      await audit(u.name, `Ticket ${p.id} → ${p.status}`, 'Support');
   402→   402→      break;
   403→   403→    }
   404→   404→
   405→   405→    /* ── Staff & super administration ── */
   406→   406→    case 'setStaffRole': {
   407→   407→      const actorRow = await db.user.findUnique({ where: { id: u.id } });
   408→   408→      if (!actorRow?.isSuperAdmin) throw new ActionError('Réservé aux super administrateurs.', 403);
   409→   409→      const newRole = String(p.role);
   410→   410→      if (!['admin', 'moderator', 'support', 'visitor'].includes(newRole)) throw new ActionError('Rôle cible invalide.', 400);
   411→   411→      const targetUser = await db.user.findUnique({ where: { id: String(p.userId) } });
   412→   412→      if (!targetUser) throw new ActionError('Compte introuvable.', 404);
   413→   413→      if (targetUser.id === u.id && newRole !== 'admin') throw new ActionError('Vous ne pouvez pas retirer vos propres accès.', 400);
   414→   414→      if (targetUser.isSuperAdmin && newRole === 'visitor') {
   415→   415→        const supers = await db.user.count({ where: { isSuperAdmin: true } });
   416→   416→        if (supers <= 1) throw new ActionError('Impossible de retirer le dernier super administrateur.', 400);
   417→   417→      }
   418→   418→      await db.user.update({
   419→   419→        where: { id: targetUser.id },
   420→   420→        data: newRole === 'visitor' ? { role: 'visitor', isSuperAdmin: false } : { role: newRole },
   421→   421→      });
   422→   422→      await audit(u.name, `Rôle staff : ${targetUser.email} → ${newRole}`, 'Administration');
   423→   423→      await notify(targetUser.id, 'Vos accès ont été mis à jour', newRole === 'visitor' ? 'Votre accès staff a été retiré par un super administrateur.' : `Votre nouveau rôle : ${newRole}.`, newRole === 'visitor' ? 'warning' : 'success');
   424→   424→      break;
   425→   425→    }
   426→   426→    case 'setSuperAdmin': {
   427→   427→      const actorRow = await db.user.findUnique({ where: { id: u.id } });
   428→   428→      if (!actorRow?.isSuperAdmin) throw new ActionError('Réservé aux super administrateurs.', 403);
   429→   429→      if (String(p.userId) === u.id) throw new ActionError('Vous ne pouvez pas modifier votre propre statut super admin.', 400);
   430→   430→      const targetUser = await db.user.findUnique({ where: { id: String(p.userId) } });
   431→   431→      if (!targetUser) throw new ActionError('Compte introuvable.', 404);
   432→   432→      const on = !!p.on;
   433→   433→      if (!on) {
   434→   434→        const supers = await db.user.count({ where: { isSuperAdmin: true } });
   435→   435→        if (supers <= 1) throw new ActionError('Impossible de retirer le dernier super administrateur.', 400);
   436→   436→      }
   437→   437→      await db.user.update({
   438→   438→        where: { id: targetUser.id },
   439→   439→        data: { isSuperAdmin: on, ...(on && !['admin', 'moderator', 'support'].includes(targetUser.role) ? { role: 'admin' } : {}) },
   440→   440→      });
   441→   441→      await audit(u.name, `Super admin ${on ? 'accordé' : 'retiré'} : ${targetUser.email}`, 'Administration');
   442→   442→      await notify(targetUser.id, on ? 'Statut super administrateur accordé' : 'Statut super administrateur retiré', on ? 'Un super administrateur vous a nommé super administrateur.' : 'Votre statut de super administrateur a été retiré.', on ? 'success' : 'warning');
   443→   443→      break;
   444→   444→    }
   445→   445→
   446→   446→    /* ── Données de démonstration & sécurité ── */
   447→   447→    case 'loadDemoData': {
   448→   448→      if (u.role !== 'admin') throw new ActionError('Réservé aux administrateurs.', 403);
   449→   449→      const users = await ensureDemoUsers(db);
   450→   450→      const content = await ensureDemoContent(db, users.userIds);
   451→   451→      await audit(u.name, `Données de démo chargées (+${users.created.length} compte(s), contenu complété)`, 'Administration');
   452→   452→      return { ok: true, accountsCreated: users.created.length, accountsTotal: users.total, content };
   453→   453→    }
   454→   454→
   455→   455→    case 'setDemoAccounts': {
   456→   456→      const actorRow = await db.user.findUnique({ where: { id: u.id } });
   457→   457→      if (!actorRow?.isSuperAdmin) throw new ActionError('Réservé aux super administrateurs.', 403);
   458→   458→      const on = !!p.on;
   459→   459→      const demoCount = await db.user.count({ where: { email: { endsWith: DEMO_DOMAIN } } });
   460→   460→      if (demoCount === 0) throw new ActionError('Aucun compte de démonstration en base — chargez d’abord les données de démo.', 400);
   461→   461→      // Les mots de passe démo sont remplacement par des hashes aléatoires : la
   462→   462→      // connexion par mot de passe devient impossible, l’accès passe uniquement
   463→   463→      // par /api/demo-login (session visiteur) quand le mode est ouvert.
   464→   464→      const hash = lockedHash();
   465→   465→      const demoRows = await db.user.findMany({ where: { email: { endsWith: DEMO_DOMAIN } }, select: { id: true } });
   466→   466→      for (const row of demoRows) {
   467→   467→        await db.user.update({ where: { id: row.id }, data: { passwordHash: hash } });
   468→   468→      }
   469→   469→      await audit(u.name, on ? `Comptes démo : ACCÈS OUVERT (${demoCount} comptes)` : 'Comptes démo : ACCÈS FERMÉ', 'Sécurité');
   470→   470→      return { ok: true, demoAccountsActive: on, accounts: demoCount };
   471→   471→    }
   472→   472→
   473→   473→    case 'resetUserPassword': {
   474→   474→      const actorRow = await db.user.findUnique({ where: { id: u.id } });
   475→   475→      if (!actorRow?.isSuperAdmin) throw new ActionError('Réservé aux super administrateurs.', 403);
   476→   476→      const targetUser = await db.user.findUnique({ where: { id: String(p.userId ?? '') } });
   477→   477→      if (!targetUser) throw new ActionError('Compte introuvable.', 404);
   478→   478→      if (targetUser.id === u.id) throw new ActionError('Pour votre propre compte, utilisez le changement de mot de passe personnel.', 400);
   479→   479→      const pwd = String(p.password ?? '');
   480→   480→      if (pwd.length < 8) throw new ActionError('Mot de passe trop court : 8 caractères minimum.', 400);
   481→   481→      await db.user.update({ where: { id: targetUser.id }, data: { passwordHash: bcrypt.hashSync(pwd, 10) } });
   482→   482→      await audit(u.name, `Mot de passe réinitialisé : ${targetUser.email}`, 'Sécurité');
   483→   483→      await notify(targetUser.id, 'Mot de passe réinitialisé', 'Un super administrateur a défini un nouveau mot de passe pour votre compte. Contactez l’équipe si vous n’êtes pas à l’origine de cette demande.', 'warning');
   484→   484→      return { ok: true, email: targetUser.email };
   485→   485→    }
   486→   486→
   487→   487→    case 'changeOwnPassword': {
   488→   488→      const me = await db.user.findUnique({ where: { id: u.id } });
   489→   489→      if (!me) throw new ActionError('Compte introuvable.', 404);
   490→   490→      if (!bcrypt.compareSync(String(p.current ?? ''), me.passwordHash)) throw new ActionError('Mot de passe actuel incorrect.', 403);
   491→   491→      const next = String(p.next ?? '');
   492→   492→      if (next.length < 8) throw new ActionError('Nouveau mot de passe trop court : 8 caractères minimum.', 400);
   493→   493→      await db.user.update({ where: { id: me.id }, data: { passwordHash: bcrypt.hashSync(next, 10) } });
   494→   494→      await audit(me.name, 'Mot de passe personnel modifié', 'Sécurité');
   495→   495→      return { ok: true };
   496→   496→    }
   497→   497→
   498→   498→    /* ── Paramètres plateforme ── */
   499→   499→    case 'updateSettings': {
   500→   500→      if (u.role !== 'admin') throw new ActionError('Réservé à l\u2019administration.', 403);
   501→   501→      const data: Record<string, unknown> = {};
   502→   502→      const keys = ['brandName', 'tagline', 'accentColor', 'maintenance', 'newRegistrations', 'autoVerifyPlayers', 'maxFreeApplications', 'featuredSlots'];
   503→   503→      for (const k of keys) if (k in p) data[k] = p[k];
   504→   504→      await db.platformSettings.upsert({
   505→   505→        where: { id: 'singleton' }, update: data,
   506→   506→        create: { id: 'singleton', ...data },
   507→   507→      });
   508→   508→      await audit(u.name, `Paramètres mis à jour : ${Object.keys(data).join(', ')}`, 'Administration');
   509→   509→      break;
   510→   510→    }
   511→   511→
   512→   512→    default:
   513→   513→      throw new ActionError(`Action inconnue : ${action}`, 404);
   514→   514→  }
   515→   515→
   516→   516→  return { ok: true };
   517→   517→}
