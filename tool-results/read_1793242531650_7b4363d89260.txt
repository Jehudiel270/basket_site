    22→/** Serialise un patch : tableaux/objets → JSON, booléens/nombres/texte → brut. */
    23→function serializePatch(patch: Record<string, unknown>, allowed: string[]): Record<string, unknown> {
    24→  const out: Record<string, unknown> = {};
    25→  for (const k of allowed) {
    26→    if (!(k in patch)) continue;
    27→    const v = patch[k];
    28→    out[k] = (Array.isArray(v) || (v && typeof v === 'object')) ? J(v) : v;
    29→  }
    30→  return out;
    31→}

    59→export async function applyAction(user: SessionUser | null, action: string, payload: Record<string, never> | any): Promise<{ ok: true; [key: string]: unknown }> {
    60→  const u = user;
    61→  const p = (payload ?? {}) as Record<string, any>;
    62→
    63→  /* ── Actions publiques (sans session) ── */
    64→  if (action === 'submitReport') {
    65→    await db.report.create({ data: {
    66→      id: `rp-${Date.now()}`, subject: String(p.reason ?? 'Signalement'), target: String(p.target ?? ''),
    67→      reason: String(p.reason ?? ''), details: String(p.details ?? ''), status: 'À traiter',
    68→      priority: 'Normale', submitted: 'À l\u2019instant', reporter: u?.name ?? 'Visiteur',

    74→
    75→  if (!u) throw new ActionError('Session requise.', 401);
    76→  const isStaff = staff(u);
    77→
    78→  switch (action) {
    79→    /* ── Joueurs ── */
    80→    case 'updatePlayer': {
    81→      if (typeof p.id !== 'string' || !p.id) throw new ActionError('Identifiant joueur manquant.', 400);
    82→      const pl = await db.player.findUnique({ where: { id: p.id } });
    83→      if (!pl || (pl.ownerId !== u.id && !isStaff)) throw new ActionError('Profil non autorisé.', 403);

    86→    }

    89→      const stats = JSON.parse(pl.stats || '[]');

    91→      await db.player.update({ where: { id: pl.id }, data: { stats: J(stats) } });
    92→      break;
    93→    }
    94→
    95→    /* ── Opportunités ── */
    96→    case 'createOpportunity': {
    97→      const org = await myOrg(u.id);
    98→      await db.opportunity.create({ data: {
    99→        id: String(p.id ?? `o-${Date.now()}`), orgId: org.id, org: org.name,

   101→        city: String(p.city ?? org.city), position: String(p.position ?? ''), ageRange: String(p.ageRange ?? ''),
   102→        level: String(p.level ?? ''), deadline: String(p.deadline ?? ''), startDate: String(p.startDate ?? ''),
   103→        status: 'Ouverte', description: String(p.description ?? ''),
   104→        requirements: J(p.requirements), conditions: J(p.conditions),
   105→        slots: Number(p.slots ?? 1), applicants: 0, salary: String(p.salary ?? ''), createdAt: today(),
   106→      } });
   107→      await audit(u.name, `Publication de l\u2019opportunité « ${p.title} »`, 'Opportunités');
   108→      break;
   109→    }
   110→    case 'updateOpportunity': {
   111→      const opp = await db.opportunity.findUnique({ where: { id: p.id } });
   112→      if (!opp) throw new ActionError('Opportunité introuvable.', 404);
   113→      const org = await db.organization.findUnique({ where: { id: opp.orgId } });
   114→      if (!org || (org.ownerId !== u.id && !isStaff)) throw new ActionError('Opportunité non autorisée.', 403);
   115→      await db.opportunity.update({ where: { id: p.id }, data: serializePatch(p.patch, OPP_FIELDS) });
   116→      break;
   117→    }
   118→
   119→    /* ── Candidatures ── */
   120→    case 'applyToOpportunity': {
   121→      const pl = await myPlayer(u.id);

   123→      if (!opp) throw new ActionError('Opportunité introuvable.', 404);

   125→      if (dup) break;

   127→        id: `a-${Date.now()}`, playerId: pl.id, playerName: pl.name, opportunityId: opp.id,
   128→        opportunityTitle: opp.title, orgId: opp.orgId, status: 'Nouveau', updatedAt: 'À l\u2019instant',
   129→        motivation: String(p.motivation ?? ''), history: J([{ status: 'Nouveau', date: today() }]),

   189→      else await db.favorite.create({ data: key });
   190→      break;
   191→    }

   194→      if (await db.follow.findUnique({ where: { userId_orgId: key } })) await db.follow.delete({ where: { userId_orgId: key } });
   195→      else await db.follow.create({ data: key });

   364→      await db.user.update({ where: { id: u.id }, data: { subStatus: 'Expiré', subRenews: '—' } });
   365→      await audit(u.name, 'Résiliation d\u2019abonnement', 'Facturation');
   366→      break;
   367→    case 'updatePlan': {
   368→      if (u.role !== 'admin') throw new ActionError('Réservé à l\u2019administration.', 403);
   369→      await db.plan.update({ where: { id: String(p.id) }, data: serializePatch(p.patch, PLAN_FIELDS) }).catch(() => {});
   370→      await audit(u.name, `Modification du plan ${p.id}`, 'Administration');
   371→      break;
   372→    }
   373→    case 'resolvePayment': {
   374→      if (!isStaff) throw new ActionError('Réservé au support.', 403);
   375→      await db.payment.update({ where: { id: String(p.id) }, data: { status: String(p.status) } }).catch(() => {});
   376→      await audit(u.name, `Paiement ${p.id} → ${p.status}`, 'Facturation');
   377→      break;
   378→    }
   379→
   380→    /* ── Support ── */
   381→    case 'createTicket': {
   382→      await db.ticket.create({ data: {
   383→        id: `t-${Date.now()}`, subject: String(p.subject ?? 'Demande'), requester: u.name,
   384→        topic: String(p.topic ?? ''), status: 'Ouvert', priority: 'Normale', opened: 'À l\u2019instant',
   385→        messages: J([{ from: 'user', body: String(p.body ?? ''), time: 'À l\u2019instant' }]),
   386→      } });
   387→      await audit(u.name, `Nouveau ticket : ${p.subject}`, 'Support');
   388→      break;
   389→    }
   390→    case 'replyTicket': {
   391→      const t = await db.ticket.findUnique({ where: { id: p.id } });
   392→      if (!t || (t.requester !== u.name && !isStaff)) throw new ActionError('Ticket non autorisé.', 403);
   393→      const messages = JSON.parse(t.messages || '[]');
   394→      messages.push({ from: isStaff && t.requester !== u.name ? 'support' : 'user', body: String(p.body ?? ''), time: 'À l\u2019instant' });
   395→      await db.ticket.update({ where: { id: t.id }, data: { messages: J(messages), status: isStaff && t.requester !== u.name ? 'En cours' : t.status } });
   396→      break;
   397→    }
   398→    case 'setTicketStatus': {
   399→      if (!isStaff) throw new ActionError('Réservé au support.', 403);
   400→      await db.ticket.update({ where: { id: String(p.id) }, data: { status: String(p.status) } }).catch(() => {});
   401→      await audit(u.name, `Ticket ${p.id} → ${p.status}`, 'Support');
   402→      break;
   403→    }
   404→
   405→    /* ── Staff & super administration ── */
   406→    case 'setStaffRole': {
   407→      const actorRow = await db.user.findUnique({ where: { id: u.id } });
   408→      if (!actorRow?.isSuperAdmin) throw new ActionError('Réservé aux super administrateurs.', 403);
   409→      const newRole = String(p.role);
   410→      if (!['admin', 'moderator', 'support', 'visitor'].includes(newRole)) throw new ActionError('Rôle cible invalide.', 400);
   411→      const targetUser = await db.user.findUnique({ where: { id: String(p.userId) } });
   412→      if (!targetUser) throw new ActionError('Compte introuvable.', 404);
   413→      if (targetUser.id === u.id && newRole !== 'admin') throw new ActionError('Vous ne pouvez pas retirer vos propres accès.', 400);
   414→      if (targetUser.isSuperAdmin && newRole === 'visitor') {
   415→        const supers = await db.user.count({ where: { isSuperAdmin: true } });
   416→        if (supers <= 1) throw new ActionError('Impossible de retirer le dernier super administrateur.', 400);
   417→      }
   418→      await db.user.update({
   419→        where: { id: targetUser.id },
   420→        data: newRole === 'visitor' ? { role: 'visitor', isSuperAdmin: false } : { role: newRole },
   421→      });
   422→      await audit(u.name, `Rôle staff : ${targetUser.email} → ${newRole}`, 'Administration');
   423→      await notify(targetUser.id, 'Vos accès ont été mis à jour', newRole === 'visitor' ? 'Votre accès staff a été retiré par un super administrateur.' : `Votre nouveau rôle : ${newRole}.`, newRole === 'visitor' ? 'warning' : 'success');
   424→      break;
   425→    }
   426→    case 'setSuperAdmin': {
   427→      const actorRow = await db.user.findUnique({ where: { id: u.id } });
   428→      if (!actorRow?.isSuperAdmin) throw new ActionError('Réservé aux super administrateurs.', 403);
   429→      if (String(p.userId) === u.id) throw new ActionError('Vous ne pouvez pas modifier votre propre statut super admin.', 400);
   430→      const targetUser = await db.user.findUnique({ where: { id: String(p.userId) } });
   431→      if (!targetUser) throw new ActionError('Compte introuvable.', 404);
   432→      const on = !!p.on;
   433→      if (!on) {
   434→        const supers = await db.user.count({ where: { isSuperAdmin: true } });
   435→        if (supers <= 1) throw new ActionError('Impossible de retirer le dernier super administrateur.', 400);
   436→      }
   437→      await db.user.update({
   438→        where: { id: targetUser.id },
   439→        data: { isSuperAdmin: on, ...(on && !['admin', 'moderator', 'support'].includes(targetUser.role) ? { role: 'admin' } : {}) },
   440→      });
   441→      await audit(u.name, `Super admin ${on ? 'accordé' : 'retiré'} : ${targetUser.email}`, 'Administration');
   442→      await notify(targetUser.id, on ? 'Statut super administrateur accordé' : 'Statut super administrateur retiré', on ? 'Un super administrateur vous a nommé super administrateur.' : 'Votre statut de super administrateur a été retiré.', on ? 'success' : 'warning');
   443→      break;
   444→    }
   445→
   446→    /* ── Données de démonstration & sécurité ── */
   447→    case 'loadDemoData': {
   448→      if (u.role !== 'admin') throw new ActionError('Réservé aux administrateurs.', 403);
   449→      const users = await ensureDemoUsers(db);
   450→      const content = await ensureDemoContent(db, users.userIds);
   451→      await audit(u.name, `Données de démo chargées (+${users.created.length} compte(s), contenu complété)`, 'Administration');
   452→      return { ok: true, accountsCreated: users.created.length, accountsTotal: users.total, content };
   453→    }
   454→
   455→    case 'setDemoAccounts': {
   456→      const actorRow = await db.user.findUnique({ where: { id: u.id } });
   457→      if (!actorRow?.isSuperAdmin) throw new ActionError('Réservé aux super administrateurs.', 403);
   458→      const on = !!p.on;
   459→      const demoCount = await db.user.count({ where: { email: { endsWith: DEMO_DOMAIN } } });
   460→      if (demoCount === 0) throw new ActionError('Aucun compte de démonstration en base — chargez d’abord les données de démo.', 400);
   461→      // Les mots de passe démo sont remplacement par des hashes aléatoires : la
   462→      // connexion par mot de passe devient impossible, l’accès passe uniquement
   463→      // par /api/demo-login (session visiteur) quand le mode est ouvert.
   464→      const hash = lockedHash();
   465→      const demoRows = await db.user.findMany({ where: { email: { endsWith: DEMO_DOMAIN } }, select: { id: true } });
   466→      for (const row of demoRows) {
   467→        await db.user.update({ where: { id: row.id }, data: { passwordHash: hash } });
   468→      }
   469→      await audit(u.name, on ? `Comptes démo : ACCÈS OUVERT (${demoCount} comptes)` : 'Comptes démo : ACCÈS FERMÉ', 'Sécurité');
   470→      return { ok: true, demoAccountsActive: on, accounts: demoCount };
   471→    }
   472→
   473→    case 'resetUserPassword': {
   474→      const actorRow = await db.user.findUnique({ where: { id: u.id } });
   475→      if (!actorRow?.isSuperAdmin) throw new ActionError('Réservé aux super administrateurs.', 403);
   476→      const targetUser = await db.user.findUnique({ where: { id: String(p.userId ?? '') } });
   477→      if (!targetUser) throw new ActionError('Compte introuvable.', 404);
   478→      if (targetUser.id === u.id) throw new ActionError('Pour votre propre compte, utilisez le changement de mot de passe personnel.', 400);
   479→      const pwd = String(p.password ?? '');
   480→      if (pwd.length < 8) throw new ActionError('Mot de passe trop court : 8 caractères minimum.', 400);
   481→      await db.user.update({ where: { id: targetUser.id }, data: { passwordHash: bcrypt.hashSync(pwd, 10) } });
   482→      await audit(u.name, `Mot de passe réinitialisé : ${targetUser.email}`, 'Sécurité');
   483→      await notify(targetUser.id, 'Mot de passe réinitialisé', 'Un super administrateur a défini un nouveau mot de passe pour votre compte. Contactez l’équipe si vous n’êtes pas à l’origine de cette demande.', 'warning');
   484→      return { ok: true, email: targetUser.email };
   485→    }
   486→
   487→    case 'changeOwnPassword': {
   488→      const me = await db.user.findUnique({ where: { id: u.id } });
   489→      if (!me) throw new ActionError('Compte introuvable.', 404);
   490→      if (!bcrypt.compareSync(String(p.current ?? ''), me.passwordHash)) throw new ActionError('Mot de passe actuel incorrect.', 403);
   491→      const next = String(p.next ?? '');
   492→      if (next.length < 8) throw new ActionError('Nouveau mot de passe trop court : 8 caractères minimum.', 400);
   493→      await db.user.update({ where: { id: me.id }, data: { passwordHash: bcrypt.hashSync(next, 10) } });
   494→      await audit(me.name, 'Mot de passe personnel modifié', 'Sécurité');
   495→      return { ok: true };
   496→    }
   497→
   498→    /* ── Paramètres plateforme ── */
   499→    case 'updateSettings': {
   500→      if (u.role !== 'admin') throw new ActionError('Réservé à l\u2019administration.', 403);
   501→      const data: Record<string, unknown> = {};
   502→      const keys = ['brandName', 'tagline', 'accentColor', 'maintenance', 'newRegistrations', 'autoVerifyPlayers', 'maxFreeApplications', 'featuredSlots'];
   503→      for (const k of keys) if (k in p) data[k] = p[k];
   504→      await db.platformSettings.upsert({
   505→        where: { id: 'singleton' }, update: data,
   506→        create: { id: 'singleton', ...data },
   507→      });
   508→      await audit(u.name, `Paramètres mis à jour : ${Object.keys(data).join(', ')}`, 'Administration');
   509→      break;
   510→    }
   511→
   512→    default:
   513→      throw new ActionError(`Action inconnue : ${action}`, 404);
   514→  }
   515→
   516→  return { ok: true };
   517→}
